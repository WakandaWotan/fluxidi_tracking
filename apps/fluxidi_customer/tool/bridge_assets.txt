# Assets the standalone customer app bundles under its own asset root, so the
# asset keys the existing customer screens already use ('assets/fluxidi/...')
# keep resolving. The golden app declares these in the repo-root pubspec, but a
# Flutter package can only share assets that live under its lib/, so the
# customer app carries its own copy.
#
# Format:
#   dir  <path relative to repo root>          copies the directory recursively
#   file <path relative to repo root>          copies one file
#
# Refresh with: dart run tool/sync_bridge_assets.dart   (from apps/fluxidi_customer)
